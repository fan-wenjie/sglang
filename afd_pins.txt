# The CUDA wheels must agree with each other or CCCL refuses the pair at JIT time with
# "CUDA compiler and CUDA toolkit headers are incompatible" -- which reads as a broken install
# and is a version skew between two wheels. It happened twice: once on the first install, and
# again when `pip install accelerate` quietly pulled nvidia-cuda-runtime back from 13.3 to 13.0.
#
# Reapply after ANY pip install in this environment:
#     python -m pip install -r afd_pins.txt
# and check with:
#     grep -m1 CUDART_VERSION $CU13/include/cuda_runtime_api.h    # must read 13030
nvidia-cuda-runtime==13.3.*
nvidia-cuda-nvrtc==13.3.*
nvidia-cuda-cupti==13.3.*
nvidia-cuda-nvcc==13.3.*
nvidia-cuda-crt==13.3.*
